<program>            ::= <statement_list>
<statement_list>     ::= <statement> { <statement> }
<statement>          ::= <declaration> | <assignment> | <conditional> | <loop>

<declaration>        ::= "sayı" <identifier> ";"
<assignment>         ::= <identifier> "=" <expr> ";"

<conditional>        ::= "eğer" "(" <condition> ")" "{" <statement_list> "}" [ "değilse" "{" <statement_list> "}" ]
<loop>               ::= "iken" "(" <condition> ")" "{" <statement_list> "}"

<condition>          ::= <expr> <relational_op> <expr>
<relational_op>      ::= "<" | ">" | "=="

<expr>               ::= <term> { ("+" | "-") <term> }
<term>               ::= <factor> { ("*" | "/") <factor> }
<factor>             ::= <identifier> | <int_literal> | "(" <expr> ")"

<identifier>         ::= <letter> { <letter> | <digit> }
<int_literal>        ::= <digit> { <digit> }

<letter> ::= ? Java Character.isLetter(char) ? | "_"
<digit>  ::= ? Java Character.isDigit(char) ?
